cmake_minimum_required(VERSION 3.20)
project(messenger_practice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CSPICE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/cspice" CACHE PATH "CSPICE installation")
add_library(cspice STATIC IMPORTED)
set_target_properties(cspice PROPERTIES IMPORTED_LOCATION "${CSPICE_ROOT}/lib/cspice.lib" INTERFACE_INCLUDE_DIRECTORIES "${CSPICE_ROOT}/include")
file(GLOB SOFA_SOURCES CONFIGURE_DEPENDS "../../vendor/sofa/*.c")
list(FILTER SOFA_SOURCES EXCLUDE REGEX "t_sofa_c\\.c$")
if(SOFA_SOURCES)
  add_library(sofa STATIC ${SOFA_SOURCES})
  target_include_directories(sofa PUBLIC ../../vendor/sofa)
endif()
add_executable(messenger src/main.cpp)
target_link_libraries(messenger PRIVATE cspice)
if(MSVC)
  target_compile_options(messenger PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_test(NAME numerical_checks COMMAND messenger selftest)
set_tests_properties(numerical_checks PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
