cmake_minimum_required(VERSION 3.20)
project(messenger_practice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MESSENGER_STAGE 4)
set(MESSENGER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(CSPICE_ROOT "${MESSENGER_ROOT}/vendor/cspice" CACHE PATH "CSPICE installation")
add_library(cspice STATIC IMPORTED)
if(WIN32)
  set(CSPICE_LIBRARY "${CSPICE_ROOT}/lib/cspice.lib")
else()
  find_library(CSPICE_LIBRARY NAMES cspice cspice.a PATHS "${CSPICE_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
endif()
set_target_properties(cspice PROPERTIES IMPORTED_LOCATION "${CSPICE_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CSPICE_ROOT}/include")
if(MESSENGER_STAGE GREATER_EQUAL 2)
  file(GLOB SOFA_SOURCES CONFIGURE_DEPENDS "${MESSENGER_ROOT}/vendor/sofa/*.c")
  list(FILTER SOFA_SOURCES EXCLUDE REGEX "t_sofa_c\\.c$")
  # Недостающие в common-data-v1 функции из того же официального выпуска SOFA.
  file(GLOB SOFA_SUPPLEMENT CONFIGURE_DEPENDS "${MESSENGER_ROOT}/third_party/sofa/*.c")
  foreach(source IN LISTS SOFA_SUPPLEMENT)
    get_filename_component(name "${source}" NAME)
    if(NOT EXISTS "${MESSENGER_ROOT}/vendor/sofa/${name}")
      list(APPEND SOFA_SOURCES "${source}")
    endif()
  endforeach()
  add_library(sofa STATIC ${SOFA_SOURCES})
  target_include_directories(sofa PUBLIC "${MESSENGER_ROOT}/vendor/sofa")
endif()
add_executable(messenger src/main.cpp)
target_compile_definitions(messenger PRIVATE MESSENGER_STAGE=${MESSENGER_STAGE})
target_link_libraries(messenger PRIVATE cspice)
option(MESSENGER_USE_SHTOOLS "Build the supplied grav.cpp against an installed SHTOOLS" OFF)
if(MESSENGER_USE_SHTOOLS)
  find_path(SHTOOLS_INCLUDE_DIR shtools/shtools.h REQUIRED)
  find_library(SHTOOLS_LIBRARY NAMES SHTOOLS shtools REQUIRED)
  target_sources(messenger PRIVATE "${MESSENGER_ROOT}/data/grav.cpp")
  target_include_directories(messenger PRIVATE "${MESSENGER_ROOT}/data" "${SHTOOLS_INCLUDE_DIR}")
  target_link_libraries(messenger PRIVATE "${SHTOOLS_LIBRARY}")
  target_compile_definitions(messenger PRIVATE MESSENGER_USE_SHTOOLS _USE_MATH_DEFINES)
endif()
if(MESSENGER_STAGE GREATER_EQUAL 2)
  target_link_libraries(messenger PRIVATE sofa)
endif()
add_executable(core_tests tests/core_tests.cpp)
add_executable(libration_tests tests/libration_tests.cpp)
if(MSVC)
  target_compile_options(messenger PRIVATE /W4 /utf-8)
  target_compile_options(core_tests PRIVATE /W4 /utf-8)
  target_compile_options(libration_tests PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_test(NAME calculated_libration_coefficients COMMAND libration_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/libration_reference.csv")
add_test(NAME numerical_checks COMMAND messenger selftest)
add_test(NAME rk8_eih_rotation COMMAND core_tests)
add_test(NAME full_degree160_field COMMAND core_tests "${MESSENGER_ROOT}/data/HgM009.sha")
set_tests_properties(numerical_checks PROPERTIES WORKING_DIRECTORY "${MESSENGER_ROOT}")
if(MESSENGER_STAGE GREATER_EQUAL 2)
  add_executable(physics_tests tests/physics_tests.cpp)
  target_link_libraries(physics_tests PRIVATE cspice sofa)
  if(MSVC)
    target_compile_options(physics_tests PRIVATE /W4 /utf-8)
  endif()
  add_test(NAME reference_rotation_and_interpolation COMMAND physics_tests "${MESSENGER_ROOT}/data")
endif()
if(MESSENGER_STAGE GREATER_EQUAL 3)
  add_executable(stage3_tests tests/stage3_tests.cpp)
  target_link_libraries(stage3_tests PRIVATE cspice sofa)
  if(MSVC)
    target_compile_options(stage3_tests PRIVATE /W4 /utf-8)
  endif()
  add_test(NAME observations_and_ramps COMMAND stage3_tests)
endif()
